// ---------- Guestbook Wishes Logic (Auto-Move Carousel) ---------- const defaultWishes = []; // Start empty — wishes persist in localStorage function getStoredWishes() { const saved = localStorage.getItem('wedding_wishes_mohamed_ashrakat'); if (saved) { try { const parsed = JSON.parse(saved); if (Array.isArray(parsed)) return parsed; } catch(e) {} } return defaultWishes; } function saveWishes(wishes) { localStorage.setItem('wedding_wishes_mohamed_ashrakat', JSON.stringify(wishes)); } function escapeHtml(str) { if (!str) return ''; return str.replace(/&/g, '&').replace(//g, '>').replace(/"/g, '"'); } let autoScrollTimer = null; let isHovered = false; function startAutoScroll() { if (autoScrollTimer) clearInterval(autoScrollTimer); autoScrollTimer = setInterval(() => { const track = document.getElementById('wishes-track'); if (!track || isHovered) return; const maxScroll = track.scrollWidth - track.clientWidth; if (maxScroll <= 0) return; if (track.scrollLeft >= maxScroll - 10) { track.scrollTo({ left: 0, behavior: 'smooth' }); } else { track.scrollBy({ left: 300, behavior: 'smooth' }); } }, 3200); } function renderWishes() { const wishes = getStoredWishes(); const track = document.getElementById('wishes-track'); const badge = document.getElementById('wishes-count-badge'); if (!track) return; if (badge) { badge.textContent = `${wishes.length} ${wishes.length === 1 ? 'Wish' : 'Wishes'}`; } if (wishes.length === 0) { track.innerHTML = `
♥ Be the first to leave a warm message for Mohamed & Ashrakat!
`; if (autoScrollTimer) clearInterval(autoScrollTimer); return; } track.innerHTML = wishes.map(w => `
${escapeHtml(w.name)}
${escapeHtml(w.relation || 'GUEST')}
"${escapeHtml(w.message)}"
${w.stars || '★★★★★ 5.0'} ${escapeHtml(w.date || 'Recently')}
`).join(''); startAutoScroll(); } // Mouse / touch pause for carousel const trackEl = document.getElementById('wishes-track'); if (trackEl) { trackEl.addEventListener('mouseenter', () => isHovered = true); trackEl.addEventListener('mouseleave', () => isHovered = false); trackEl.addEventListener('touchstart', () => isHovered = true, {passive: true}); trackEl.addEventListener('touchend', () => isHovered = false, {passive: true}); } // Form submission const wishForm = document.getElementById('wish-form'); if (wishForm) { wishForm.addEventListener('submit', (e) => { e.preventDefault(); const nameInput = document.getElementById('guest-name'); const relationInput = document.getElementById('guest-relation'); const messageInput = document.getElementById('guest-message'); const toast = document.getElementById('wish-success-toast'); const name = nameInput.value.trim(); const relation = relationInput.value; const message = messageInput.value.trim(); if (!name || !message) return; const currentWishes = getStoredWishes(); const newWish = { name, relation, message, stars: "★★★★★ 5.0", date: "Just now" }; currentWishes.unshift(newWish); saveWishes(currentWishes); renderWishes(); wishForm.reset(); if (toast) { toast.style.display = 'block'; setTimeout(() => { toast.style.display = 'none'; }, 4000); } if (trackEl) { trackEl.scrollTo({ left: 0, behavior: 'smooth' }); } }); } // Carousel buttons const prevBtn = document.getElementById('wish-prev'); const nextBtn = document.getElementById('wish-next'); if (prevBtn && trackEl) { prevBtn.addEventListener('click', () => { trackEl.scrollBy({ left: -320, behavior: 'smooth' }); }); } if (nextBtn && trackEl) { nextBtn.addEventListener('click', () => { trackEl.scrollBy({ left: 320, behavior: 'smooth' }); }); } renderWishes();
